// Optional PBR materials (Shabby + Resurfaced), so the hull matches SEP's engines under Deferred. Without them the parts
// use the stock specular shader.
SHABBY_MATERIAL_DEF:NEEDS[Resurfaced]
{
	name = ITS_Hull_PBR
	shader = Resurfaced/Standard
	FLOAT
	{
		_Smoothness = 1.0
		_Metalness = 0.0
		_MetalAlbedoMultiplier = 1.0
	}
}
SHABBY_MATERIAL_DEF:NEEDS[Resurfaced]
{
	name = ITS_Details_PBR
	shader = Resurfaced/Standard
	TEXTURE
	{
		_MetalMap = StarshipThingyMabobsAndSuch/Assets/ITS/its_det_metal
	}
	FLOAT
	{
		_Smoothness = 1.0
		_Metalness = 1.0
		_MetalAlbedoMultiplier = 1.15
	}
}

{

	{
		materialDef = ITS_Details_PBR

		materialDef = ITS_Hull_PBR
